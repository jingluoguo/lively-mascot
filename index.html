✦ lively-mascot

通用电子宠物引擎：40 种状态表情 · 多角色 · 纯 SVG 驱动 · 零依赖 · 即插即用。

把光标移到吉祥物上 · 点击让它开心

// 任意容器内创建
var mascot = LivelyMascot.createMascot(el, {
  type: 'sprout',  // 或 'cat'
  color: '#48ff42', size: 180,
  onClick: () => console.log('hi'),
});

// 切换表情
mascot.setEmotion('10');  // Happy
mascot.setEmotion('38');  // Eureka

// 切换角色
LivelyMascot.registerCharacter('myChar', renderFn, 'MyChar');
mascot = LivelyMascot.createMascot(el, { type: 'myChar' });
